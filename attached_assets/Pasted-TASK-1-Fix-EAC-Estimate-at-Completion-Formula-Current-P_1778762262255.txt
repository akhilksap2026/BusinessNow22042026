Step 1.1: Create new file lib/eac.ts
typescriptexport interface TaskEAC {
  actualHours: number;
  estimateHours: number;
  completionPct: number;
  eacHours: number;
  varianceHours: number;
  status: 'under' | 'on-track' | 'over';
}

export function calculateEAC(
  actualHours: number,
  estimateHours: number,
  completionPct: number
): TaskEAC {
  // If no work done yet, EAC = estimate
  if (completionPct === 0 || actualHours === 0) {
    return {
      actualHours,
      estimateHours,
      completionPct,
      eacHours: estimateHours,
      varianceHours: 0,
      status: 'on-track'
    };
  }

  // Calculate Estimate To Complete (ETC)
  const etc = actualHours * ((1 - completionPct) / completionPct);
  const eacHours = actualHours + etc;
  const varianceHours = eacHours - estimateHours;

  // Determine status
  let status: 'under' | 'on-track' | 'over';
  const variancePct = estimateHours > 0 ? (varianceHours / estimateHours) : 0;
  
  if (variancePct < -0.05) {
    status = 'under';
  } else if (variancePct > 0.05) {
    status = 'over';
  } else {
    status = 'on-track';
  }

  return {
    actualHours,
    estimateHours,
    completionPct,
    eacHours: Math.round(eacHours * 100) / 100, // 2 decimal places
    varianceHours: Math.round(varianceHours * 100) / 100,
    status
  };
}
Step 1.2: Create test file lib/eac.test.ts
typescriptimport { describe, it, expect } from 'vitest';
import { calculateEAC } from './eac';

describe('calculateEAC', () => {
  it('returns estimate when no progress', () => {
    const result = calculateEAC(0, 10, 0);
    expect(result.eacHours).toBe(10);
    expect(result.varianceHours).toBe(0);
    expect(result.status).toBe('on-track');
  });

  it('calculates overrun correctly', () => {
    // 8 hours spent, 50% complete, estimate 10
    // ETC = 8 * (0.5 / 0.5) = 8
    // EAC = 8 + 8 = 16
    const result = calculateEAC(8, 10, 0.5);
    expect(result.eacHours).toBe(16);
    expect(result.varianceHours).toBe(6);
    expect(result.status).toBe('over');
  });

  it('calculates underrun correctly', () => {
    // 5 hours spent, 50% complete, estimate 10
    // ETC = 5 * (0.5 / 0.5) = 5
    // EAC = 5 + 5 = 10
    const result = calculateEAC(5, 10, 0.5);
    expect(result.eacHours).toBe(10);
    expect(result.varianceHours).toBe(0);
    expect(result.status).toBe('on-track');
  });

  it('handles 100% completion', () => {
    const result = calculateEAC(12, 10, 1.0);
    // At 100%, EAC should equal actuals
    expect(result.eacHours).toBe(12);
    expect(result.varianceHours).toBe(2);
  });
});
