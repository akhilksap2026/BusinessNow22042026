1.1.1 Apply per-table cascade rules

Table.Column	FK Target	Strategy	Rationale
tasks.project_id	projects.id	CASCADE	Tasks have no meaning without parent project
tasks.parent_task_id	tasks.id	SET NULL	Promote orphan subtasks to top-level
task_notes.task_id	tasks.id	CASCADE	Notes are dependent records

time_entries.task_id	tasks.id	SET NULL	Same
time_entries.timesheet_id	timesheets.id	SET NULL	Same
allocations.project_id	projects.id	CASCADE	Allocations are scoped to project
allocations.user_id	users.id	SET NULL	Convert to placeholder allocation
invoices.project_id	projects.id	RESTRICT	Block project deletion if invoiced (financial trail)
invoices.account_id	accounts.id	RESTRICT	Same

Phase 4 — Security Hardening
4.1 helmet
app.use(helmet({ contentSecurityPolicy: false }));  // CSP audit deferred

4.2 Dual-key rate limiting
// Per-IP — distributed brute force defence
app.use("/api", rateLimit({ windowMs: 60_000, max: 600, keyGenerator: (req) => req.ip }));
// Per-user — single-account abuse defence
app.use("/api", rateLimit({
  windowMs: 60_000,
  max: 200,
  keyGenerator: (req) => String(req.headers["x-user-id"] ?? req.ip),
  skip: (req) => !req.headers["x-user-id"],
}));

4.3 Global error handler + process-level catchers
// app.ts (last middleware)
app.use((err, _req, res, _next) => {
  logger.error({ err }, "Unhandled error");
  res.status(500).json({ error: "Internal server error" });
});
// index.ts
process.on("unhandledRejection", (reason) => { logger.error({ reason }, "Unhandled rejection"); process.exit(1); });
process.on("uncaughtException", (err) => { logger.error({ err }, "Uncaught exception"); process.exit(1); });
